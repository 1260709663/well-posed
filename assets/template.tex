\section*{示例：定理盒的写法}
\addcontentsline{toc}{section}{示例：定理盒的写法}

\begin{thmbox}{定理 1.2 \refo{(Theorem 1.2, p.4)}}
设 $\Omega\subset\mathbb{R}^d$ 为有界光滑区域，$u_0\in H^1_0(\Omega)$。
则对任意 $T>0$ 存在唯一弱解 $u\in L^\infty(0,T;H^1_0)\cap L^2(0,T;H^2)$，且
\begin{equation}
\|u(t)\|_{H^1}^2+\int_0^t\|u_s(s)\|^2\,ds\le C\|u_0\|_{H^1}^2 .
\end{equation}
\end{thmbox}

\begin{proofbox}{补证：从 (3.7) 到 (3.9) 的能量估计}
原文写 ``by standard energy estimates''。具体做法是取 $u_s$ 作测试函数……
\begin{align}
\tfrac12\frac{d}{dt}\|\nabla u\|^2+\|u_s\|^2
&= -\int \nabla p\cdot\nabla u_s \;\le\; \tfrac12\|u_s\|^2 + C\|\nabla p\|^2 .
\end{align}
再对 $t$ 积分并用 Gronwall 不等式即得。
\end{proofbox}

\begin{gapbox}{跳步清单条目示例}
\refo{p.7, l.12}：\emph{``a density argument''} —— 需要在 $C_c^\infty$ 上先证，再用 $\|\cdot\|_{W^{1,\infty}}$ 的弱下半连续性取极限；注意此处需 $d\le 4$ 才能保证嵌入。
\end{gapbox}

\begin{warnbox}{不确定处示例}
作者引用的 \refo{[12]} 中的定理要求 $\partial\Omega\in C^2$，而本文只假设 Lipschitz；两者是否兼容，原文未说明。
\end{warnbox}
